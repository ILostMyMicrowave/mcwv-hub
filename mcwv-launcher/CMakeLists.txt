cmake_minimum_required(VERSION 3.20)
project(MCWV-Launcher VERSION 1.0.1 LANGUAGES CXX RC)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
add_executable(MCWV-Launcher WIN32 src/main-clean.cpp src/version.rc)
target_link_libraries(MCWV-Launcher PRIVATE winhttp crypt32 shell32 kernel32 user32)
if(MSVC)
    set_property(TARGET MCWV-Launcher PROPERTY MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
    target_link_options(MCWV-Launcher PRIVATE /MANIFEST:EMBED /MANIFESTINPUT:${CMAKE_SOURCE_DIR}/src/app.manifest)
endif()
